cmake_minimum_required(VERSION 3.24)
# Source configuration only: a toolchain and every dependency must be supplied
# and independently verified by the qualification process. No discovery,
# download, installation or runtime qualification is performed here.
if(NOT DEFINED CMAKE_TOOLCHAIN_FILE OR NOT IS_ABSOLUTE "${CMAKE_TOOLCHAIN_FILE}")
  message(FATAL_ERROR "An explicit qualified toolchain file is required")
endif()
project(pro273_primitives LANGUAGES CXX)
option(PRO273_INSTALLER_ROOT "Configure the unqualified Windows installer root target" OFF)
if(PRO273_INSTALLER_ROOT)
  if(NOT WIN32)
    message(FATAL_ERROR "The installer root target requires Windows")
  endif()
  add_executable(pro273_installer_root pro273_primitives.cpp)
  target_compile_features(pro273_installer_root PRIVATE cxx_std_17)
  target_compile_definitions(pro273_installer_root PRIVATE PRO273_INSTALLER_ROOT UNICODE _UNICODE WIN32_LEAN_AND_MEAN)
  target_link_libraries(pro273_installer_root PRIVATE advapi32 bcrypt)
  return()
endif()
set(required_inputs PRO273_PG_SERVER_INCLUDE PRO273_PG_SOURCE_ROOT PRO273_ICU_INCLUDE PRO273_ICU_SOURCE_ROOT PRO273_ICU_UC_LIBRARY PRO273_ICU_I18N_LIBRARY PRO273_ICU_DATA_ROOT PRO273_DOUBLE_CONVERSION_INCLUDE PRO273_DOUBLE_CONVERSION_SOURCE_ROOT PRO273_DOUBLE_CONVERSION_LIBRARY)
# Linux extensions resolve PostgreSQL symbols from the loading backend.
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  list(APPEND required_inputs PRO273_PG_LIBRARY)
endif()
foreach(required IN LISTS required_inputs)
  if(NOT DEFINED ${required} OR NOT IS_ABSOLUTE "${${required}}" OR NOT EXISTS "${${required}}")
    message(FATAL_ERROR "Explicit existing absolute ${required} is required")
  endif()
endforeach()
add_library(pro273_primitives MODULE pro273_primitives.cpp)
target_compile_features(pro273_primitives PRIVATE cxx_std_17)
target_include_directories(pro273_primitives PRIVATE "${PRO273_PG_SERVER_INCLUDE}" "${PRO273_ICU_INCLUDE}" "${PRO273_DOUBLE_CONVERSION_INCLUDE}")
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  target_link_libraries(pro273_primitives PRIVATE "${PRO273_PG_LIBRARY}")
endif()
target_link_libraries(pro273_primitives PRIVATE "${PRO273_ICU_UC_LIBRARY}" "${PRO273_ICU_I18N_LIBRARY}" "${PRO273_DOUBLE_CONVERSION_LIBRARY}")
if(MSVC)
  target_compile_options(pro273_primitives PRIVATE /fp:strict /EHs-c-)
else()
  target_compile_options(pro273_primitives PRIVATE -fno-fast-math -ffp-contract=off -frounding-math -fno-exceptions)
endif()
set_target_properties(pro273_primitives PROPERTIES PREFIX "")
